@RestController
@RequestMapping("/api/boyaca-ai")
public class BoyacaAiController {

    private final ChatClient chatClient;
    private final VectorStore vectorStore;

    public BoyacaAiController(ChatClient.Builder b, VectorStore s) {
        this.chatClient = b.defaultSystem(
            "Asistente oficial Java & AI Day UPTC Tunja"
        ).build();
        this.vectorStore = s;
    }

    @PostMapping("/query")
    public ResponseEntity<String> ask(@RequestBody String q) {
        // Virtual Threads para baja latencia
        return ResponseEntity.ok(chatClient.prompt().user(q).call().content());
    }
}
